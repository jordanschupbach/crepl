cmake_minimum_required(VERSION 3.20)
project(crepl VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build: cmake -S . -B build/debug")
endif()
find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 21)
  message(FATAL_ERROR "crepl currently supports LLVM/Clang 21; found ${LLVM_PACKAGE_VERSION}")
endif()
find_program(CREPL_CLANG_EXECUTABLE NAMES clang-21 clang HINTS "${CLANG_INSTALL_PREFIX}/bin" REQUIRED)
execute_process(COMMAND "${CREPL_CLANG_EXECUTABLE}" -print-resource-dir
  OUTPUT_VARIABLE CREPL_RESOURCE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
# Embedding Clang bypasses compiler wrappers. Preserve the build toolchain's
# system headers, including Nix's standard library and libc directories.
set(CREPL_CPP_INCLUDES "")
foreach(dir IN LISTS CMAKE_CXX_IMPLICIT_INCLUDE_DIRECTORIES)
  string(APPEND CREPL_CPP_INCLUDES "  \"${dir}\",\n")
endforeach()
set(CREPL_C_INCLUDES "")
foreach(dir IN LISTS CMAKE_C_IMPLICIT_INCLUDE_DIRECTORIES)
  string(APPEND CREPL_C_INCLUDES "  \"${dir}\",\n")
endforeach()
set(CREPL_LIBRARY_DIRS "")
foreach(dir IN LISTS CMAKE_CXX_IMPLICIT_LINK_DIRECTORIES)
  string(APPEND CREPL_LIBRARY_DIRS "  \"${dir}\",\n")
endforeach()
configure_file(cmake/config.hpp.in generated/crepl/config.hpp @ONLY)
add_library(crepl_core source/input.cpp source/options.cpp source/session.cpp)
target_include_directories(crepl_core PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(crepl_core SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
target_compile_options(crepl_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(crepl_core PRIVATE clang-cpp LLVM)
add_executable(crepl source/main.cpp)
target_link_libraries(crepl PRIVATE crepl_core)
# Export interpreter runtime symbols used by JIT-generated expression printers.
set_target_properties(crepl PROPERTIES ENABLE_EXPORTS ON)
install(TARGETS crepl RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES README.md docs/plan.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
find_package(Doxygen QUIET)
if(DOXYGEN_FOUND)
  set(DOXYGEN_EXTRACT_ALL YES)
  set(DOXYGEN_EXTRACT_PRIVATE YES)
  set(DOXYGEN_GENERATE_XML YES)
  set(DOXYGEN_WARN_IF_UNDOCUMENTED YES)
  set(DOXYGEN_USE_MDFILE_AS_MAINPAGE "${CMAKE_CURRENT_SOURCE_DIR}/README.md")
  doxygen_add_docs(docs include source README.md docs/plan.md)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
